enable
configure terminal

! ----------------------------------------------------
! Hostname
! ----------------------------------------------------
hostname SW-A-DCEE

! ----------------------------------------------------
! VLAN Creation
! ----------------------------------------------------
vlan 10
 name VLAN_DEIE
vlan 20
 name VLAN_DCEE
vlan 30
 name VLAN_DMME
vlan 40
 name VLAN_DIS
vlan 99
 name MGMT
vlan 100
 name NATIVE
exit

! ----------------------------------------------------
! Switchports: Access Ports (PCs)
! ----------------------------------------------------
interface fastEthernet 0/1
 description PC-PT_PC7
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown

interface fastEthernet 0/2
 description PC-PT_PC6
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown

! ----------------------------------------------------
! Switchports: Trunk Uplink to SW-D-DCEE
! ----------------------------------------------------
interface fastEthernet 0/3
 description TRUNK_TO_SW-D-DCEE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 100
 switchport trunk allowed vlan 10,20,30,40,99,100
 no shutdown

! ----------------------------------------------------
! Management Interface and Gateway
! ----------------------------------------------------
interface vlan 99
 description MGMT_INTERFACE
 ip address 10.99.99.22 255.255.255.0
 no shutdown
exit

ip default-gateway 10.99.99.1

end
write memory
