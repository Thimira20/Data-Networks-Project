enable
configure terminal

! ----------------------------------------------------
! Hostname
! ----------------------------------------------------
hostname SW-A-DIS

! ----------------------------------------------------
! SSH & Local User Configuration
! ----------------------------------------------------
username admin privilege 15 secret admin123
ip domain-name campus.uor.lk
crypto key generate rsa general-keys modulus 1024

line vty 0 15
 transport input ssh
 login local
exit

! ----------------------------------------------------
! SNMP Configuration
! ----------------------------------------------------
snmp-server community public RO
snmp-server community private RW
snmp-server host 10.10.40.100 version 2c public
snmp-server enable traps

! ----------------------------------------------------
! VLAN Creation
! ----------------------------------------------------
vlan 10
 name VLAN_DEIE
vlan 20
 name VLAN_DCEE
vlan 30
 name VLAN_DMME
vlan 40
 name VLAN_DIS
vlan 99
 name MGMT
vlan 100
 name NATIVE
exit

! ----------------------------------------------------
! Switchports: Trunk Uplink to SW-D-DIS
! ----------------------------------------------------
interface fastEthernet 0/1
 description TRUNK_TO_SW-D-DIS
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 100
 switchport trunk allowed vlan 10,20,30,40,99,100
 no shutdown

! ----------------------------------------------------
! Switchports: Access Ports (PCs / Servers)
! ----------------------------------------------------
interface fastEthernet 0/2
 description PC-PT_PC2
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 no shutdown

interface fastEthernet 0/3
 description PC-PT_PC3
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 no shutdown

! ----------------------------------------------------
! Management Interface and Gateway
! ----------------------------------------------------
interface vlan 99
 description MGMT_INTERFACE
 ip address 10.99.99.24 255.255.255.0
 no shutdown
exit

ip default-gateway 10.99.99.1

end
write memory
