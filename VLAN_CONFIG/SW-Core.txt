enable

! ----------------------------------------------------
! SNMP Configuration

snmp-server community public RO

snmp-server host 10.10.40.100 version 2c public

! ----------------------------------------------------
! VLAN Creation
! ----------------------------------------------------
vlan 10
 name VLAN_DEIE
vlan 20
 name VLAN_DCEE
vlan 30
 name VLAN_DMME
vlan 40
 name VLAN_DIS
vlan 99
 name MGMT
vlan 100
 name NATIVE
exit

! ----------------------------------------------------
! Spanning Tree Root configuration
! ----------------------------------------------------
spanning-tree vlan 1-100 root primary

! ----------------------------------------------------
! Switchports: Trunk Uplinks to Distribution Switches
! ----------------------------------------------------
interface fastEthernet 0/1
 description TRUNK_TO_SW-D-DCEE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 100
 switchport trunk allowed vlan 10,20,30,40,99,100
 no shutdown

interface fastEthernet 0/2
 description TRUNK_TO_SW-D-DEIE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 100
 switchport trunk allowed vlan 10,20,30,40,99,100
 no shutdown

interface fastEthernet 0/3
 description TRUNK_TO_SW-D-DMME

 no shutdown

! ----------------------------------------------------
! Access Control Lists (ACLs) Design
! ----------------------------------------------------
! ACL for VLAN_DEIE (VLAN 10)
ip access-list extended ACL-DEIE-IN
 permit ip 10.10.10.0 0.0.0.255 10.10.40.0 0.0.0.255
 deny ip 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255
 deny ip 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255
 deny ip 10.10.10.0 0.0.0.255 10.99.99.0 0.0.0.255
 permit ip 10.10.10.0 0.0.0.255 any
exit

! ACL for VLAN_DCEE (VLAN 20)
ip access-list extended ACL-DCEE-IN
 permit tcp 10.10.20.0 0.0.0.255 10.10.40.0 0.0.0.255 eq 80
 permit tcp 10.10.20.0 0.0.0.255 10.10.40.0 0.0.0.255 eq 443

 deny ip 10.10.20.0 0.0.0.255 10.99.99.0 0.0.0.255
 permit ip 10.10.20.0 0.0.0.255 any
exit

! ACL for VLAN_DMME (VLAN 30) - Completely isolated
ip access-list extended ACL-DMME-IN
 deny ip 10.10.30.0 0.0.0.255 10.10.10.0 0.0.0.255
 deny ip 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255
 deny ip 10.10.30.0 0.0.0.255 10.10.40.0 0.0.0.255
 deny ip 10.10.30.0 0.0.0.255 10.99.99.0 0.0.0.255
 deny ip 10.10.30.0 0.0.0.255 any
exit

! ACL for VLAN_DIS (VLAN 40) - Servers
ip access-list extended ACL-DIS-IN
 permit udp host 10.10.40.100 10.99.99.0 0.0.0.255 eq 161
 permit udp host 10.10.40.100 10.0.0.0 0.0.0.255 eq 161
 permit udp host 10.10.40.100 10.0.1.0 0.0.0.255 eq 161
 permit icmp host 10.10.40.100 10.99.99.0 0.0.0.255
 permit icmp host 10.10.40.100 10.0.0.0 0.0.0.255
 permit icmp host 10.10.40.100 10.0.1.0 0.0.0.255
 permit ip 10.10.40.0 0.0.0.255 10.10.10.0 0.0.0.255
 permit tcp 10.10.40.0 0.0.0.255 established 10.10.20.0 0.0.0.255
 deny ip 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255
 deny ip 10.10.40.0 0.0.0.255 10.10.30.0 0.0.0.255
 deny ip 10.10.40.0 0.0.0.255 10.99.99.0 0.0.0.255
 deny ip 10.10.40.0 0.0.0.255 any
exit

 permit tcp 10.99.99.0 0.0.0.255 10.99.99.0 0.0.0.255 eq 22
 permit tcp 10.99.99.0 0.0.0.255 10.0.0.0 0.0.0.255 eq 22
